langchain-text-splitters==1.1.2

# ---------- 环境变量 ----------

python-dotenv==1.2.3

# ---------- 向量库 + 本地模型（Ollama）----------
langchain-chroma==1.1.0
chromadb==1.5.9
langchain-ollama==1.1.0

# ---------- 云端大模型 ----------
# 走任何"兼容 OpenAI 接口"的服务都用它，本项目用来接 DeepSeek。

langchain-openai==1.7.0

# ---------- Web 服务（把 Agent 包成 HTTP 接口）----------
# 对外提供 POST /ask，入口是 api.py。
# 只有跑 api.py 才需要；只跑 agent/graph.py 那个命令行入口不需要。
# 注：api.py 用 def 而不是 async def 写路由 ——
#     uvicorn 会自动把同步函数扔进线程池，多个请求能并行处理。
fastapi==0.143.0
uvicorn==0.53.0

# ---------- 可观测性（把每次提问的完整轨迹上报 Langfuse）----------
# 接线在 api.py：CallbackHandler 挂进 graph.stream 的 config["callbacks"]。
# 只有跑 api.py 才需要；只跑 agent/graph.py 那个命令行入口不需要。
#
# ⚠️ 密钥从 .env 读（LANGFUSE_PUBLIC_KEY / LANGFUSE_SECRET_KEY），
#    缺密钥不会报错 —— SDK 打印一句 "Client will be disabled" 就自己关掉，
#    服务照常运行，只是不上报。所以别人没配也能跑起来。
langfuse==4.18.0

# ⚠️⚠️ 下面这行【必须显式钉住】，别以为它是多余的 ——
#     langfuse 依赖 opentelemetry-*-proto-common 1.45.1，
#     而 chromadb 依赖 opentelemetry-exporter-otlp-proto-grpc。
#     若 grpc exporter 停在 1.44.0，两边的内部模块对不上，
#     chromadb 直接 ImportError → 整个向量检索挂掉、服务起不来。
#     实测踩过一次（2026-10-14），所以这里把版本钉死。
opentelemetry-exporter-otlp-proto-grpc==1.45.1

# ---------- PDF 解析 ----------
pymupdf==1.28.2

# ---------- 重排（cross-encoder）----------
# 只有跑 hybrid_rerank 才需要，纯向量/BM25 检索不需要。
# 这里装的是 torch 的 CPU 版（约 124 MB）；有 N 卡想加速可以换 CUDA 版，但体积大很多。
torch==2.14.1
sentence-transformers==6.1.0
